    // CameraX binding. ImageAnalysis is ALWAYS bound when the device allows it (the previous code lost
    // it whenever ViewPort creation failed on a 0x0 PreviewView), and the ViewPort is built only after
    // layout so Preview, Capture and Analysis share exactly the same field of view.
    LaunchedEffect(hasCameraPermission, cameraSelector, previewViewRef, resumedCount) {
        val pView = previewViewRef
        if (!hasCameraPermission || pView == null) return@LaunchedEffect
        try {
            awaitLaidOut(pView)
            val provider = awaitCameraProvider(context)
            cameraProvider = provider
            val bound = bindScannerCamera(
                provider = provider,
                lifecycleOwner = lifecycleOwner,
                previewView = pView,
                cameraSelector = cameraSelector,
                flashMode = flashMode,
                analysisExecutor = analysisExecutor,
                controller = scanController
            )
            scanController.reset()
            if (bound == null) {
                imageCapture = null
                cameraControl = null
                cameraInfo = null
                return@LaunchedEffect
            }
            imageCapture = bound.imageCapture
            cameraControl = bound.camera.cameraControl
            cameraInfo = bound.camera.cameraInfo
            bound.camera.cameraInfo.zoomState.observe(lifecycleOwner) { state ->
                zoomRatio = state.zoomRatio
                minZoomRatio = state.minZoomRatio
                maxZoomRatio = state.maxZoomRatio
            }
        } catch (e: kotlinx.coroutines.CancellationException) {
            throw e
        } catch (_: Throwable) {
            scanController.reset()
        }
    }

    fun capturePhoto() {
        val cap = imageCapture
        if (cap == null) {
            Toast.makeText(
                context,
                if (isArabic) "جاري تشغيل الكاميرا، يرجى الانتظار ثانية..." else "Initializing camera, please wait…",
                Toast.LENGTH_SHORT
            ).show()
            launchSystemCamera()
            return
        }
        if (isCapturing || captureCooldown) return
        isCapturing = true
        showFlashEffect = true

        // Snapshot at shutter time, read directly from the controller (never a stale composition value).
        val priorQuad = scanController.captureQuadForShutter()
        val modeAtShutter = scanMode
        scanController.onCaptureStarted()

        val cacheDir = File(context.cacheDir, "camera_scans").apply { if (!exists()) mkdirs() }
        val photoFile = File(cacheDir, "scan_${System.currentTimeMillis()}.jpg")
        val outputOptions = ImageCapture.OutputFileOptions.Builder(photoFile).build()

        cap.takePicture(
            outputOptions,
            ContextCompat.getMainExecutor(context),
            object : ImageCapture.OnImageSavedCallback {
                override fun onImageSaved(outputFileResults: ImageCapture.OutputFileResults) {
                    showFlashEffect = false
                    coroutineScope.launch {
                        // Decode + EXIF + detection(with prior) + warp + filter: all off the main thread.
                        val page = try {
                            scanController.processCapture(context, photoFile, modeAtShutter, priorQuad)
                        } catch (e: kotlinx.coroutines.CancellationException) {
                            throw e
                        } catch (e: Exception) {
                            e.printStackTrace()
                            null
                        }
                        if (page == null) {
                            isCapturing = false
                            scanController.onCaptureFinished(stayOnCamera = true)
                            Toast.makeText(
                                context,
                                if (isArabic) "تعذر معالجة الصورة، حاول مرة أخرى" else "Could not process the photo, please retry",
                                Toast.LENGTH_SHORT
                            ).show()
                            return@launch
                        }
                        triggerHapticFeedback()
                        val rawPath = page.rawPath
                        val procPath = page.processedPath

                        when (modeAtShutter) {
                            ScanCameraMode.DOCUMENT -> {
                                isCapturing = false
                                scanController.onCaptureFinished(stayOnCamera = false)
                                onDocumentCaptured(listOf(Pair(rawPath, procPath)))
                            }
                            ScanCameraMode.BATCH -> {
                                batchPages.add(Pair(rawPath, procPath))
                                isCapturing = false
                                captureCooldown = true
                                scanController.onCaptureFinished(stayOnCamera = true)
                                delay(800)
                                captureCooldown = false
                            }
                            ScanCameraMode.PASSPORT -> {
                                if (!isPassportFrontDone) {
                                    passportFrontPath = procPath
                                    passportFrontRawPath = rawPath
                                    isPassportFrontDone = true
                                    isCapturing = false
                                    captureCooldown = true
                                    scanController.onCaptureFinished(stayOnCamera = true)
                                    delay(800)
                                    captureCooldown = false
                                } else {
                                    isCapturing = false
                                    scanController.onCaptureFinished(stayOnCamera = false)
                                    onDocumentCaptured(
                                        listOf(
                                            Pair(passportFrontRawPath ?: rawPath, passportFrontPath ?: procPath),
                                            Pair(rawPath, procPath)
                                        )
                                    )
                                }
                            }
                            ScanCameraMode.ID_CARD -> {
                                if (!isIdCardFrontDone) {
                                    idCardFrontPath = procPath
                                    idCardFrontRawPath = rawPath
                                    isIdCardFrontDone = true
                                    isCapturing = false
                                    captureCooldown = true
                                    scanController.onCaptureFinished(stayOnCamera = true)
                                    delay(800)
                                    captureCooldown = false
                                } else {
                                    isCapturing = false
                                    scanController.onCaptureFinished(stayOnCamera = false)
                                    onDocumentCaptured(
                                        listOf(
                                            Pair(idCardFrontRawPath ?: rawPath, idCardFrontPath ?: procPath),
                                            Pair(rawPath, procPath)
                                        )
                                    )
                                }
                            }
                        }
                    }
                }

                override fun onError(exception: ImageCaptureException) {
                    isCapturing = false
                    showFlashEffect = false
                    scanController.onCaptureFinished(stayOnCamera = true)
                    launchSystemCamera()
                }
            }
        )
    }
