    private var importJob: kotlinx.coroutines.Job? = null

    fun setImportedUrisPendingEdit(uris: List<Uri>, autoCrop: Boolean = true) {
        if (uris.isEmpty()) return
        importJob?.cancel()
        _uiState.update {
            it.copy(
                importedUrisPending = uris,
                pagesPendingEdit = emptyList(),
                pendingPages = emptyList(),
                isLoading = true,
                isBackgroundProcessing = true
            )
        }
        importJob = viewModelScope.launch {
            val completed = mutableListOf<Pair<String, String>>()
            for ((index, uri) in uris.withIndex()) {
                try {
                    // Same pipeline as camera / batch: decode + EXIF + detection + warp + filter.
                    val page = DocumentPipeline.processUri(context, uri, autoCrop = autoCrop, prefix = "import_p${index + 1}")
                    if (page != null) {
                        completed += Pair(page.rawPath, page.processedPath)
                        _uiState.update {
                            it.copy(
                                pagesPendingEdit = completed.toList(),
                                pendingPages = completed.toList(),
                                isLoading = false // first page ready -> Edit is usable now
                            )
                        }
                        if (completed.size == 1) analyzePendingFirstPage()
                    } else {
                        _events.send(UiEvent.Error("Could not import page ${index + 1}"))
                    }
                } catch (e: kotlinx.coroutines.CancellationException) {
                    throw e
                } catch (e: Exception) {
                    e.printStackTrace()
                    _events.send(UiEvent.Error("Failed to process page ${index + 1}"))
                }
            }
            _uiState.update { it.copy(isLoading = false, isBackgroundProcessing = false) }
        }
    }

    fun autoCropAllPendingPages() {
        val currentPages = _uiState.value.pagesPendingEdit
        if (currentPages.isEmpty()) return
        viewModelScope.launch {
            _uiState.update { it.copy(isBackgroundProcessing = true) }
            try {
                for (index in currentPages.indices) {
                    val (rawPath, oldProcessed) = currentPages[index]
                    val page = DocumentPipeline.redetect(context, rawPath, prefix = "recrop_p${index + 1}") ?: continue
                    if (page.processedPath.isNotBlank() && page.processedPath != oldProcessed) {
                        // Update by raw path, not index: pages may have been reordered/deleted meanwhile.
                        _uiState.update { s ->
                            val updated = s.pagesPendingEdit.map { if (it.first == rawPath) Pair(rawPath, page.processedPath) else it }
                            s.copy(pagesPendingEdit = updated, pendingPages = updated)
                        }
                    }
                }
            } catch (e: kotlinx.coroutines.CancellationException) {
                throw e
            } catch (e: Exception) {
                e.printStackTrace()
                _events.send(UiEvent.Error("Failed to auto-crop pending pages"))
            } finally {
                _uiState.update { it.copy(isBackgroundProcessing = false) }
            }
        }
    }

// Shared helper (add once, inside the class):
    /** Stored quad for a page: DB json -> sidecar -> full image. Never the old 6% inset. */
    private fun quadFor(rawPath: String, cropQuadJson: String): DocumentQuad =
        DocumentQuad.fromJsonOrNull(cropQuadJson) ?: QuadStore.load(rawPath) ?: DocumentQuad.fullQuad()

    fun batchAutoCropAllSessionPages() {
        val pages = if (_uiState.value.isEditingSession) _uiState.value.editingSessionPages else _uiState.value.activePages
        if (pages.isEmpty()) return

        viewModelScope.launch {
            _uiState.update { it.copy(isLoading = true) }
            try {
                for (page in pages) {
                    val rawPath = page.rawImagePath.ifBlank { page.processedImagePath }
                    val keepFilter = runCatching { FilterType.valueOf(page.filterType) }.getOrDefault(DocumentPipeline.DEFAULT_FILTER)
                    val result = DocumentPipeline.redetect(
                        context, rawPath,
                        rotationDegrees = page.rotationDegrees,
                        filter = keepFilter,
                        prefix = "batch_recrop"
                    ) ?: continue
                    val updated = page.copy(
                        processedImagePath = result.processedPath,
                        cropQuadJson = result.quad.toJson(),
                        filterType = keepFilter.name
                    )
                    if (_uiState.value.isEditingSession) {
                        _uiState.update { s -> s.copy(editingSessionPages = s.editingSessionPages.map { if (it.id == page.id) updated else it }) }
                    } else {
                        repository.updatePage(updated)
                        _uiState.update { s -> s.copy(activePages = s.activePages.map { if (it.id == page.id) updated else it }) }
                    }
                    qualityCache.remove(page.id)
                }
            } catch (e: kotlinx.coroutines.CancellationException) {
                throw e
            } catch (e: Exception) {
                e.printStackTrace()
                _events.send(UiEvent.Error("Failed to auto-crop pages"))
            } finally {
                _uiState.update { it.copy(isLoading = false) }
                val docId = _uiState.value.activeDocument?.id
                if (docId != null && !_uiState.value.isEditingSession) loadDocument(docId)
            }
        }
    }

    fun applyFilterToActivePage(filter: FilterType) {
        val pages = _uiState.value.activePages
        val idx = _uiState.value.selectedPageIndex
        if (idx !in pages.indices) return
        val page = pages[idx]

        viewModelScope.launch {
            _uiState.update { it.copy(isLoading = true) }
            try {
                val newPath = DocumentPipeline.render(
                    context,
                    page.rawImagePath,
                    quadFor(page.rawImagePath, page.cropQuadJson),
                    page.rotationDegrees,
                    filter,
                    prefix = "filter_active"
                ) ?: throw IllegalStateException("Unable to load original page")
                repository.updatePage(page.copy(processedImagePath = newPath, filterType = filter.name))
                qualityCache.remove(page.id)
                loadDocument(page.documentId)
            } catch (e: kotlinx.coroutines.CancellationException) {
                throw e
            } catch (e: Exception) {
                e.printStackTrace()
                _events.send(UiEvent.Error("Failed to apply filter"))
            } finally {
                _uiState.update { it.copy(isLoading = false) }
            }
        }
    }

    fun rotatePage(pageId: Long) {
        viewModelScope.launch {
            _uiState.update { it.copy(isLoading = true) }
            try {
                val page = repository.getPageById(pageId) ?: return@launch
                val newRotation = ((page.rotationDegrees + 90) % 360 + 360) % 360
                val filter = runCatching { FilterType.valueOf(page.filterType) }.getOrDefault(DocumentPipeline.DEFAULT_FILTER)
                // Same order as the crop editor: warp(raw, quad) -> rotate -> filter.
                val newPath = DocumentPipeline.render(
                    context,
                    page.rawImagePath,
                    quadFor(page.rawImagePath, page.cropQuadJson),
                    newRotation,
                    filter,
                    prefix = "rotate"
                ) ?: throw IllegalStateException("Unable to load original page")
                repository.updatePage(page.copy(processedImagePath = newPath, rotationDegrees = newRotation))
                qualityCache.remove(pageId)
                loadDocument(page.documentId)
            } catch (e: kotlinx.coroutines.CancellationException) {
                throw e
            } catch (e: Exception) {
                e.printStackTrace()
                _events.send(UiEvent.Error("Failed to rotate page"))
            } finally {
                _uiState.update { it.copy(isLoading = false) }
            }
        }
    }
